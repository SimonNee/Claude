================================================================================
  std::vector QUICK REFERENCE
================================================================================

HEADER:  #include <vector>

--------------------------------------------------------------------------------
  CONSTRUCTION
--------------------------------------------------------------------------------

  vector<int> v;                      // empty vector
  vector<int> v(5);                   // 5 elements, default-initialised (0)
  vector<int> v(5, 42);               // 5 elements, all 42
  vector<int> v = {1, 2, 3, 4, 5};   // initialiser list
  vector<int> v(other);               // copy constructor
  vector<int> v(other.begin(), other.end()); // range constructor

--------------------------------------------------------------------------------
  SIZE & CAPACITY
--------------------------------------------------------------------------------

  v.size()        // number of elements (size_t)
  v.empty()       // true if size == 0
  v.capacity()    // allocated storage (>= size)
  v.max_size()    // theoretical maximum elements

  v.resize(n)     // grow/shrink to n elements (new ones default-init'd)
  v.resize(n, x)  // grow/shrink to n elements (new ones set to x)
  v.reserve(n)    // pre-allocate capacity for n elements (no size change)
  v.shrink_to_fit() // release excess capacity

--------------------------------------------------------------------------------
  ELEMENT ACCESS
--------------------------------------------------------------------------------

  v[i]            // access element (no bounds check, UB if out of range)
  v.at(i)         // access element (throws std::out_of_range if invalid)
  v.front()       // first element
  v.back()        // last element
  v.data()        // raw pointer to underlying array

--------------------------------------------------------------------------------
  MODIFIERS
--------------------------------------------------------------------------------

  v.push_back(x)          // append element (copy)
  v.emplace_back(args...) // append element constructed in-place (prefer this)
  v.pop_back()            // remove last element

  v.insert(it, x)         // insert x before iterator it
  v.insert(it, n, x)      // insert n copies of x before it
  v.insert(it, first, last) // insert range before it
  v.emplace(it, args...)  // construct element in-place before it

  v.erase(it)             // remove element at iterator
  v.erase(first, last)    // remove range [first, last)

  v.clear()               // remove all elements (capacity unchanged)
  v.assign(n, x)          // replace contents with n copies of x
  v.assign(first, last)   // replace contents with range
  v.swap(other)           // swap contents with another vector

--------------------------------------------------------------------------------
  ITERATORS
--------------------------------------------------------------------------------

  v.begin()   / v.end()     // forward iterators
  v.rbegin()  / v.rend()    // reverse iterators
  v.cbegin()  / v.cend()    // const forward iterators
  v.crbegin() / v.crend()   // const reverse iterators

  // Range-based for (preferred)
  for (const auto& x : v) { ... }
  for (auto& x : v)       { ... }  // modifying

--------------------------------------------------------------------------------
  COMMON ALGORITHMS (with vector)
--------------------------------------------------------------------------------

  #include <algorithm>

  std::sort(v.begin(), v.end());                    // sort ascending
  std::sort(v.begin(), v.end(), std::greater<int>{}); // sort descending
  std::reverse(v.begin(), v.end());
  std::find(v.begin(), v.end(), x);                // returns iterator or end()
  std::count(v.begin(), v.end(), x);               // count occurrences
  std::accumulate(v.begin(), v.end(), 0);          // sum (#include <numeric>)
  std::min_element(v.begin(), v.end());            // iterator to min
  std::max_element(v.begin(), v.end());            // iterator to max
  std::fill(v.begin(), v.end(), x);               // set all to x
  std::transform(v.begin(), v.end(), out.begin(), fn); // apply fn to each
  std::remove(v.begin(), v.end(), x);             // "remove" (erase-remove idiom)

  // Erase-remove idiom (actually removes elements)
  v.erase(std::remove(v.begin(), v.end(), x), v.end());

  // Remove duplicates (sort first)
  std::sort(v.begin(), v.end());
  v.erase(std::unique(v.begin(), v.end()), v.end());

--------------------------------------------------------------------------------
  COMPARISON
--------------------------------------------------------------------------------

  v1 == v2   // element-wise equality
  v1 != v2
  v1 < v2    // lexicographic comparison
  v1 > v2

--------------------------------------------------------------------------------
  MEMORY / PERFORMANCE NOTES
--------------------------------------------------------------------------------

  - push_back / emplace_back: amortised O(1), but may reallocate
  - reserve() upfront avoids reallocations if size is known
  - emplace_back() constructs in-place; prefer over push_back for objects
  - insert/erase in the middle: O(n) - elements must shift
  - Accessing by index: O(1)
  - Contiguous in memory - cache-friendly, compatible with C arrays via data()
